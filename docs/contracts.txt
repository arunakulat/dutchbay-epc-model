@dataclass(frozen=True)
class ShockSpec:
    """
    [SENS-001] Specification for a single variable shock.
    CCCDIR Compliant: Immutable, Typed, Validated.
    """
    variable_name: str  # e.g., "project.capex_usd_per_kw"
    base_value: float   # e.g., 1200.0
    low_pct: float      # e.g., -10.0
    high_pct: float     # e.g., +10.0
    label: str | None = None

    def __post_init__(self) -> None:
        """CESSPIT Fail-Fast Validation."""
        if self.base_value <= 0:
            raise ValueError(f"ShockSpec: base_value must be > 0, got {self.base_value}")
        if self.low_pct > 0:
            raise ValueError(f"ShockSpec: low_pct must be <= 0, got {self.low_pct}")
        if self.high_pct < 0:
            raise ValueError(f"ShockSpec: high_pct must be >= 0, got {self.high_pct}")
        if self.high_pct < abs(self.low_pct):
            # Optional warning logic could go here, but strict check is safer
            pass

    @property
    def low_value(self) -> float:
        return self.base_value * (1 + self.low_pct / 100.0)

    @property
    def high_value(self) -> float:
        return self.base_value * (1 + self.high_pct / 100.0)

    @property
    def display_label(self) -> str:
        return self.label or self.variable_name

@dataclass(frozen=True)
class ShockResult:
    """
    [SENS-001] Result of a single shock applied to a metric.
    Stores absolute values AND computed deltas for Tornado charts.
    """
    variable_name: str
    metric_name: str
    base_value: float    # Input parameter base
    low_value: float     # Input parameter low
    high_value: float    # Input parameter high

    base_metric: float   # Output metric base (e.g., IRR 12%)
    low_metric: float    # Output metric at low input
    high_metric: float   # Output metric at high input

    label: str | None = None

    @property
    def impact(self) -> float:
        """Absolute magnitude of change (High - Low)."""
        return abs(self.high_metric - self.low_metric)

    @property
    def direction(self) -> Literal["positive", "negative", "neutral", "mixed"]:
        """Direction of correlation between Input and Output."""
        delta_input = self.high_value - self.low_value
        delta_metric = self.high_metric - self.low_metric

        if abs(delta_metric) < 1e-9:
            return "neutral"

        # If input increases and metric increases -> positive correlation
        return "positive" if (delta_metric * delta_input) > 0 else "negative"

    @property
    def sensitivity(self) -> float:
        """Elasticity: % change in metric / % change in input."""
        if self.base_value == 0 or self.base_metric == 0:
            return 0.0

        pct_change_input = (self.high_value - self.low_value) / self.base_value
        pct_change_metric = (self.high_metric - self.low_metric) / self.base_metric

        if abs(pct_change_input) < 1e-9:
            return 0.0

        return pct_change_metric / pct_change_input

@dataclass(frozen=True)
class SensitivitySuite:
    """
    [SENS-001] Collection of ShockResults for a scenario (Tornado).
    """
    metric_name: str
    base_metric_value: float
    scenario_name: str
    shock_results: List[ShockResult]
    analysis_timestamp: str = field(default_factory=lambda: datetime.now().isoformat())
    base_config_path: Optional[str] = None
    metadata: Dict[str, Any] = field(default_factory=dict)

    @property
    def tornado_ranking(self) -> List[ShockResult]:
        """Return shocks sorted by impact (descending)."""
        return sorted(self.shock_results, key=lambda x: x.impact, reverse=True)

    @property
    def metric(self) -> str:
        return self.metric_name

    @property
    def base_metric(self) -> float:
        return self.base_metric_value

    @property
    def tornado_results(self) -> List[ShockResult]:
        return self.tornado_ranking

class StandardShockLibrary:
    """
    [SENS-001] Lender-Grade Standard Shocks (Factory).
    Returns ready-to-use ShockSpec objects.
    """
    @staticmethod
    def capex_overrun(base: float, pct: float = 10.0) -> ShockSpec:
        return ShockSpec("project.capex_usd_per_kw", base, -pct, pct, "CAPEX Sensitivity")

    @staticmethod
    def opex_variation(base: float, pct: float = 10.0) -> ShockSpec:
        return ShockSpec("project.opex_usd_per_kw", base, -pct, pct, "OPEX Sensitivity")

    @staticmethod
    def capacity_factor(base: float, pct: float = 10.0) -> ShockSpec:
        return ShockSpec("project.capacity_factor", base, -pct, pct, "Generation (CF)")

    @staticmethod
    def power_price(base: float, pct: float = 15.0) -> ShockSpec:
        return ShockSpec("project.tariff_usd_per_kwh", base, -pct, pct, "Tariff / Price")

    @staticmethod
    def fx_usd_lkr(base: float, pct: float = 10.0) -> ShockSpec:
        # Note: FX shock logic might need specific handling if struct used
        return ShockSpec("finance.fx.lkr_per_usd", base, -pct, pct, "FX (USD/LKR)")

    @staticmethod
    def debt_tenor(base: float, pct: float = 20.0) -> ShockSpec:
        return ShockSpec("finance.debt.tenor_years", base, -pct, pct, "Debt Tenor")

    @staticmethod
    def interest_rate(base: float, bps: int = 200) -> ShockSpec:
        # 200 bps = 2.0%
        pct_impact = (bps / 10000.0) / base * 100.0 if base > 0 else 0.0
        # For rates, we typically add/subtract bps, but ShockSpec uses pct of base.
        # This wrapper calculates the pct needed to achieve the bps shift.
        # Logic assumes base is e.g. 0.08 (8%).
        return ShockSpec("finance.debt.interest_rate", base, -pct_impact, pct_impact, f"Interest Rate (+/- {bps}bps)")

# 5. CASPER Payload Helper
# ═════════════════════════════════════════════════════════════════════════════

def build_casper_payload(
    *,
    scenario: ScenarioResult,
    baseline_kpis: Mapping[str, float] | None = None,
    sensitivities: SensitivitySuite | None = None,
    monte_carlo: MonteCarloResult | None = None,
    multi_tech_generation_breakdown: MultiTechGenerationResult | None = None,
    technology_breakdown: Sequence[TechnologyBreakdown] | None = None,
    metadata: Mapping[str, Any] | None = None,
    tail_risk_snapshots: Mapping[str, TailRiskSnapshot] | None = None,
) -> Dict[str, Any]:
    """
    Build a JSON-safe CASPER payload.
    """
    payload = {
        "contract_version": CASPER_CONTRACT_VERSION,
        "scenario": scenario.as_dict(),
        "baseline_kpis": dict(baseline_kpis) if baseline_kpis else {},
    }

    if sensitivities:
        tornado_rows: list[Dict[str, Any]] = []
        for row in sensitivities.tornado_results:
            impact_abs = abs(row.high_metric - row.low_metric)
            impact_pct = (impact_abs / row.base_metric * 100.0) if row.base_metric != 0 else 0.0
            tornado_rows.append({
                "variable": row.variable_name,
                "base_irr": float(row.base_metric),
                "low_irr": float(row.low_metric),
                "high_irr": float(row.high_metric),
                "impact_abs": float(impact_abs),
                "impact_pct": float(impact_pct),
            })
        payload["sensitivity"] = {
            "metric": sensitivities.metric,
            "base_metric": float(sensitivities.base_metric),
            "base_config_path": sensitivities.base_config_path,
            "tornado": tornado_rows,
        }

    if monte_carlo:
        payload["monte_carlo"] = {
            "iterations": int(monte_carlo.iterations),
            "failed_iterations": int(monte_carlo.failed_iterations),
            "success_rate_pct": monte_carlo.success_rate(),
            "irr": {
                "mean": float(monte_carlo.project_irr_mean),
                "std": float(monte_carlo.project_irr_std),
                "p10": float(monte_carlo.project_irr_p10),
                "p50": float(monte_carlo.project_irr_p50),
                "p90": float(monte_carlo.project_irr_p90),
                "se": float(monte_carlo.project_irr_se),
            },
            "npv": {
                "mean": float(monte_carlo.project_npv_mean),
                "p10": float(monte_carlo.project_npv_p10),
                "p50": float(monte_carlo.project_npv_p50),
                "p90": float(monte_carlo.project_npv_p90),
                "se": float(monte_carlo.project_npv_se),
            },
            "dscr_min": {
                "p10": float(monte_carlo.dscr_min_p10),
                "p50": float(monte_carlo.dscr_min_p50),
                "se": float(monte_carlo.dscr_min_se),
            },
        }

    if multi_tech_generation_breakdown:
        payload["generation"] = multi_tech_generation_breakdown.to_dict()

    if technology_breakdown:
        payload["technology_breakdown"] = [asdict(tb) for tb in technology_breakdown]

    if tail_risk_snapshots:
        tail_risk_block: Dict[str, Any] = {}
        for metric, snapshot in tail_risk_snapshots.items():
            tail_risk_block[metric] = {
                "metric": snapshot.metric,
                "confidence": float(snapshot.confidence),
                "var": float(snapshot.var),
                "cvar": float(snapshot.cvar),
                "p10": float(snapshot.p10),
                "p50": float(snapshot.p50),
                "p90": float(snapshot.p90),
                "breach_probability": float(snapshot.breach_probability),
            }
        if tail_risk_block:
            payload["tail_risk"] = tail_risk_block

    payload["metadata"] = dict(metadata) if metadata else {}

    return payload
